Migrating from JSON to Multipart Form Data: Frontend Guide

Overview

This document provides guidance on migrating frontend requests from application/json content type to multipart/form-data to support file uploads in our authentication system.

Affected Endpoints

The following endpoints now support or require multipart/form-data:

  1. POST /auth/signup - For user registration with optional student ID upload
  2. POST /auth/google - For Google authentication with optional student ID upload

Required Changes

1. Form Submission Approach

Previous (JSON):

// Using fetch with JSON
fetch('/auth/signup', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    first_name: 'John',
    last_name: 'Doe',
    email: 'john.doe@example.com',
    is_student: false
  })
});

// Using axios with JSON
axios.post('/auth/signup', {
  first_name: 'John',
  last_name: 'Doe',
  email: 'john.doe@example.com',
  is_student: false
});

New (Multipart Form Data):

// Using FormData with fetch
const formData = new FormData();
formData.append('first_name', 'John');
formData.append('last_name', 'Doe');
formData.append('email', 'john.doe@example.com');
formData.append('is_student', 'true'); // String 'true' or 'false'

// If student, append file
if (studentIdFile) {
  formData.append('file', studentIdFile);
}

fetch('/auth/signup', {
  method: 'POST',
  body: formData
});

// Using FormData with axios
axios.post('/auth/signup', formData, {
  headers: {
    'Content-Type': 'multipart/form-data'
  }
});

2. Field Requirements

Signup Endpoint

Field Type Required Description
first_name String Yes Min 2, max 50 characters, letters and spaces only
last_name String Yes Min 2, max 50 characters, letters and spaces only
email String Yes Valid email format, max 255 characters
is_student Boolean No Default: false, send as ‘true’ or ‘false’ string
student_id File Conditional Required if is_student=true, must be an image file

Google Auth Endpoint

Field Type Required Description
email String Yes Email address from Google account
first_name String Yes First name from Google account
last_name String Yes Last name from Google account
google_id String Yes Unique identifier from Google
is_student Boolean No Default: false, send as ‘true’ or ‘false’ string
student_id File Conditional Required if is_student=true, must be an image file

3. File Upload Requirements

For student ID uploads:

  • File must be an image (JPG, PNG, GIF, etc.)
  • File field name must be file
  • Maximum file size: 5MB (configured on server)
  • Required when is_student is set to true

4. Boolean Values in Form Data

When using multipart/form-data, boolean values must be sent as strings:

// Correct
formData.append('is_student', 'true');
formData.append('is_student', 'false');

// Also accepted
formData.append('is_student', '1');  // true
formData.append('is_student', '0');  // false

5. Response Handling

The response format remains the same as before, with JSON responses containing:

  • Success/error messages
  • User data (when applicable)
  • Authentication tokens (when applicable)

Implementation Examples

React Implementation Example

import { useState } from 'react';
import axios from 'axios';

function SignupForm() {
  const [formData, setFormData] = useState({
    first_name: '',
    last_name: '',
    email: '',
    is_student: false
  });
  const [file, setFile] = useState(null);
  const [error, setError] = useState(null);
  const [success, setSuccess] = useState(false);

  const handleChange = (e) => {
    const { name, value, type, checked } = e.target;
    setFormData({
      ...formData,
      [name]: type === 'checkbox' ? checked : value
    });
  };

  const handleFileChange = (e) => {
    setFile(e.target.files[0]);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError(null);
    
    const submitData = new FormData();
    submitData.append('first_name', formData.first_name);
    submitData.append('last_name', formData.last_name);
    submitData.append('email', formData.email);
    submitData.append('is_student', formData.is_student ? 'true' : 'false');
    
    if (formData.is_student && file) {
      submitData.append('file', file);
    } else if (formData.is_student && !file) {
      setError('Student ID image is required when registering as a student');
      return;
    }
    
    try {
      const response = await axios.post('/auth/signup', submitData);
      setSuccess(true);
    } catch (err) {
      setError(err.response?.data?.message || 'An error occurred during signup');
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {error && <div className="error">{error}</div>}
      {success && <div className="success">Account created! Please check your email.</div>}
      
      <div>
        <label htmlFor="first_name">First Name:</label>
        <input
          type="text"
          id="first_name"
          name="first_name"
          value={formData.first_name}
          onChange={handleChange}
          required
        />
      </div>
      
      <div>
        <label htmlFor="last_name">Last Name:</label>
        <input
          type="text"
          id="last_name"
          name="last_name"
          value={formData.last_name}
          onChange={handleChange}
          required
        />
      </div>
      
      <div>
        <label htmlFor="email">Email:</label>
        <input
          type="email"
          id="email"
          name="email"
          value={formData.email}
          onChange={handleChange}
          required
        />
      </div>
      
      <div>
        <label>
          <input
            type="checkbox"
            name="is_student"
            checked={formData.is_student}
            onChange={handleChange}
          />
          I am a student
        </label>
      </div>
      
      {formData.is_student && (
        <div>
          <label htmlFor="student_id">Student ID Image:</label>
          <input
            type="file"
            id="student_id"
            name="student_id"
            accept="image/*"
            onChange={handleFileChange}
          />
        </div>
      )}
      
      <button type="submit">Sign Up</button>
    </form>
  );
}

Google Authentication Example

import { useGoogleLogin } from '@react-oauth/google';
import axios from 'axios';

function GoogleSignupButton() {
  const [isStudent, setIsStudent] = useState(false);
  const [file, setFile] = useState(null);
  
  const handleGoogleLogin = useGoogleLogin({
    onSuccess: async (codeResponse) => {
      // First get user info from Google
      const googleUserInfo = await fetchGoogleUserInfo(codeResponse.access_token);
      
      // Then authenticate with our server
      submitGoogleAuth(googleUserInfo);
    },
    onError: (error) => {
      console.error('Google login failed:', error);
    }
  });
  
  const fetchGoogleUserInfo = async (accessToken) => {
    const response = await axios.get('https://www.googleapis.com/oauth2/v3/userinfo', {
      headers: { Authorization: `Bearer ${accessToken}` }
    });
    return response.data;
  };
  
  const submitGoogleAuth = async (googleUser) => {
    const formData = new FormData();
    formData.append('email', googleUser.email);
    formData.append('first_name', googleUser.given_name);
    formData.append('last_name', googleUser.family_name);
    formData.append('google_id', googleUser.sub);
    formData.append('is_student', isStudent ? 'true' : 'false');
    
    if (isStudent && file) {
      formData.append('file', file);
    }
    
    try {
      const response = await axios.post('/auth/google', formData, {
        headers: {
          'Content-Type': 'multipart/form-data'
        }
      });
      
      // Handle successful authentication
      console.log('Authentication successful', response.data);
      
    } catch (error) {
      console.error('Authentication error:', error);
    }
  };
  
  return (
    <div>
      <div>
        <label>
          <input
            type="checkbox"
            checked={isStudent}
            onChange={(e) => setIsStudent(e.target.checked)}
          />
          I am a student
        </label>
      </div>
      
      {isStudent && (
        <div>
          <label htmlFor="student_id">Student ID Image:</label>
          <input
            type="file"
            id="student_id"
            accept="image/*"
            onChange={(e) => setFile(e.target.files[0])}
          />
        </div>
      )}
      
      <button onClick={() => handleGoogleLogin()}>Sign in with Google</button>
    </div>
  );
}
